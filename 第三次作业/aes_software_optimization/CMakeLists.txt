cmake_minimum_required(VERSION 3.16)
project(aes_software_optimization C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(AESLAB_SANITIZE "Enable ASan/UBSan" OFF)
add_library(aeslab STATIC
    src/aes_common.c src/aes_baseline.c src/aes_ttable.c src/aes_shuffle.c
    src/aes_aesni.c src/aes_vaes.c src/modes.c)
target_include_directories(aeslab PUBLIC include PRIVATE src)
target_compile_options(aeslab PRIVATE -O3 -Wall -Wextra -Wpedantic)
add_executable(aeslab_test tests/test_aeslab.c)
target_link_libraries(aeslab_test PRIVATE aeslab)
target_compile_options(aeslab_test PRIVATE -O2 -Wall -Wextra)
add_executable(aeslab_bench bench/bench.c)
target_link_libraries(aeslab_bench PRIVATE aeslab)
target_compile_options(aeslab_bench PRIVATE -O3 -Wall -Wextra)
if(AESLAB_SANITIZE)
  foreach(t aeslab aeslab_test aeslab_bench)
    target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  endforeach()
  target_link_options(aeslab_test PRIVATE -fsanitize=address,undefined)
  target_link_options(aeslab_bench PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME aeslab_vectors COMMAND aeslab_test)
